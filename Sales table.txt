CREATE TABLE sales (
    sale_id INT PRIMARY KEY,
    customer_name VARCHAR(100),
    product VARCHAR(100),
    category VARCHAR(50),
    quantity INT,
    price DECIMAL(10,2),
    city VARCHAR(50),
    sales_date DATE,
    salesperson VARCHAR(100),
    discount DECIMAL(5,2)
);
